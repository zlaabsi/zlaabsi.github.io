{{- /* The image shown when a link to this page is shared. A post or explorable gets its own card, static/images/og/
-.png, written by scripts/generate_og_cards.py. Every other page, and any page whose card has not been generated yet, gets the site card from params.images. Returns a dict {url, alt}, or false when no image is configured. */ -}} {{- $url := "" -}} {{- $alt := site.Title -}} {{- with .Params.images -}} {{- $url = index . 0 | absURL -}} {{- end -}} {{- if and (not $url) .IsPage .File -}} {{- $card := printf "images/og/%s-%s.png" .Section .File.ContentBaseName -}} {{- if fileExists (printf "static/%s" $card) -}} {{- $url = absURL $card -}} {{- $alt = printf "%s, by %s" .Title site.Title -}} {{- end -}} {{- end -}} {{- if not $url -}} {{- with site.Params.images -}}{{- $url = index . 0 | absURL -}}{{- end -}} {{- end -}} {{- $image := false -}} {{- if $url -}}{{- $image = dict "url" $url "alt" $alt -}}{{- end -}} {{- return $image -}}
